model {
  for (t in 1:n) {
    # Verosimilitud: conteo de desapariciones ~ Poisson
    y[t] ~ dpois(mu[t])

    # Media: log-link con covariables base + offset de población
    log(mu[t]) <- beta[1] + beta[2]*homicidios[t] + beta[3]*robos[t]
                  + beta[4]*violencia[t] + beta[5]*pct_seguros[t]
                  + beta[6]*post2018[t] + offset[t]

    # Predictiva
    yf[t] ~ dpois(mu[t])
  }

  # Iniciales
  for (j in 1:6) {
    beta[j] ~ dnorm(0, 0.001)
  }
}
