model {
  for (i in 1:n) {
    # Verosimilitud: conteo de desapariciones por alcaldia x mes
    y[i] ~ dpois(mu[i])

    # Media: log-link con covariables base + efecto aleatorio por alcaldia + offset
    log(mu[i]) <- beta[1] + beta[2]*homicidios[i] + beta[3]*robos[i]
                  + beta[4]*violencia[i] + beta[5]*pct_seguros[i]
                  + beta[6]*post2018[i] + theta[alcaldia[i]] + log(offset[i])

    # Predictiva
    yf[i] ~ dpois(mu[i])
  }

  # Efectos aleatorios por alcaldia: theta_k ~ N(0, tau_b)
  for (k in 1:K) {
    theta[k] ~ dnorm(0, tau.b)
  }

  # Iniciales para betas (efectos fijos)
  for (j in 1:6) {
    beta[j] ~ dnorm(0, 0.001)
  }

  # Precision del efecto aleatorio
  tau.b ~ dgamma(0.01, 0.01)
  sig.b <- 1 / sqrt(tau.b)
}
